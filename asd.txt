ALTURA_TORRE_MM = 2000
NIVELES_TORRE = 11
ALTURA_SEGURIDAD_MM = 2000

# Alturas por nivel (en mm)
ALTURAS_Z = [int(i * ALTURA_TORRE_MM / NIVELES_TORRE) for i in range(NIVELES_TORRE)]

# Valores posibles de Y (en mm)
VALORES_Y = list(range(-500, 510, 100))

def simular_movimiento_robot(x, y, z):
    """
    Simula el movimiento del brazo KUKA en coordenadas cartesianas.
    """
    print(f"🔧 Movimiento del brazo: X={x} mm, Y={y} mm, Z={z} mm")

def mover_robot_a_altura_segura():
    """
    Simula la subida del brazo a la altura de seguridad.
    """
    simular_movimiento_robot(1000, 0, ALTURA_SEGURIDAD_MM)
    print("🔼 El brazo se posicionó en altura segura.\n")

def ciclo_de_desmoldeo():
    global variables

    idReceta_actual = 1
    RecetaNombres = [
    "(OV-A) OVALADO A",
    "(OV-B) OVALADO B",
    "(CU) CUADRADO",
    "(QP) QUESO PUERCO",
    "(7K) RECTANGULAR",
    "(6K) MANDOLINA",
    "(LU) LUNCH"
    ]
    gripper_actual = 1
    torre_actual = 1
    torre_proxima = 2
    estado_maquina = 1
    desmoldeo_banda = 1
    valor_X = 0
    
    # Buscar variables importantes
    fin_cancelado_var = None
    fin_finalizado_var = None
    estado_actual_var = None
    niveles_desmoldados_var = None
    niveles_seleccionados_var = None
    pantalla_receta_var = None
    
    for var in variables:
        nombre = var.get_browse_name().Name
        if nombre == "finCancelado":
            fin_cancelado_var = var
        elif nombre == "finFinalizado":
            fin_finalizado_var = var
        elif nombre == "Estado_actual":
            estado_actual_var = var
        elif nombre == "nivelesDesmoldados":
            niveles_desmoldados_var = var
        elif nombre == "nivelesSeleccionados":
            niveles_seleccionados_var = var
        elif nombre == "pantalla_receta":
            pantalla_receta_var = var
    
    while True:
        # Leer valores actuales configurados desde UaExpert
        if estado_actual_var:
            estado_maquina = estado_actual_var.get_value()
        
        if pantalla_receta_var:
            pantalla_receta_estado = pantalla_receta_var.get_value()
            print(f"Estado de pantalla_receta: {pantalla_receta_estado}")
        
        # Reset de flags de finalización
        if fin_cancelado_var:
            fin_cancelado_var.set_value(False)
        if fin_finalizado_var:
            fin_finalizado_var.set_value(False)
        
        # Asegurar que nivelesDesmoldados esté en False al inicio
        if niveles_desmoldados_var:
            niveles_desmoldados_var.set_value(False)
            
        receta_nombre = RecetaNombres[idReceta_actual - 1]
        print(f"\n🟢 Iniciando ciclo de desmoldeo para la receta: {receta_nombre}")

        for repeticion in range(1, 2):
            print(f"\n🔁 Repetición {repeticion}/2")
            receta_0 = recetario.get_child([f"2:[{idReceta_actual}]"])

            peso_producto_node = receta_0.get_child(["2:PESO DEL PRODUCTO"])
            
            nuevo_peso = random.uniform(14.3, 16)
            peso_producto_node.set_value(nuevo_peso)

            # Obtenemos cantidad de niveles seleccionados (por defecto 11)
            niveles_total = 11
            if niveles_seleccionados_var:
                niveles_total = niveles_seleccionados_var.get_value()
            
            # Bandera para controlar si el ciclo fue cancelado
            ciclo_cancelado = False
            
            for nivel_actual in range(1, niveles_total + 1):
                print(f"Torre {torre_actual}, Nivel {nivel_actual}")
                
                # 5% de probabilidad de fallo en el desmoldeo
                probabilidad_fallo = random.random() <= 0.05
                
                if probabilidad_fallo:
                    print("❌ ERROR: Fallo en el desmoldeo - Cancelando ciclo")
                    if fin_cancelado_var:
                        fin_cancelado_var.set_value(True)
                    # Marcar el ciclo como cancelado
                    ciclo_cancelado = True
                    # Cancelar el ciclo actual y pasar al siguiente
                    break
                
                # Obtener altura Z del nivel actual (en mm)
                altura_nivel_z = ALTURAS_Z[nivel_actual - 1] if nivel_actual <= len(ALTURAS_Z) else ALTURAS_Z[-1]
                posicion_y = random.choice(VALORES_Y)
                valor_X = 0
                
                # Simular movimiento hacia el molde
                simular_movimiento_robot(valor_X, posicion_y, altura_nivel_z)
                print("🧲 Brazo recoge el molde del nivel")

                # Actualizar variables en el servidor OPC UA
                for var in variables:
                    nombre = var.get_browse_name().Name
                    if nombre == "N_receta_actual":
                        var.set_value(idReceta_actual)
                    elif nombre == "N_receta_proxima":
                        var.set_value((idReceta_actual % len(RecetaNombres)) + 1)
                    elif nombre == "N_torre_actual":
                        var.set_value(torre_actual)
                    elif nombre == "N_torre_proxima":
                        var.set_value(torre_proxima)
                    elif nombre == "sdda_nivel_actual":
                        var.set_value(nivel_actual)
                    elif nombre == "desmoldeobanda":
                        var.set_value(desmoldeo_banda)
                    elif nombre == "posicionX":
                        var.set_value(valor_X)
                    elif nombre == "posicionY":
                        var.set_value(posicion_y)
                    elif nombre == "posicionZ":
                        var.set_value(altura_nivel_z)
                    elif nombre == "Ciclo_iniciado":
                        var.set_value(True)

                valor_X = 1000
                # Simular movimiento hacia zona de descarga
                simular_movimiento_robot(valor_X, posicion_y, altura_nivel_z)
                print("📤 Brazo deposita el molde en la zona de descarga")

                # Subida a altura de seguridad
                mover_robot_a_altura_segura()
                
                if nivel_actual == niveles_total:
                    # Primero activar nivelesDesmoldados y mantenerlo activo
                    if niveles_desmoldados_var:
                        niveles_desmoldados_var.set_value(True)
                        print(f"✅ Nivel {nivel_actual} completado - Activando flanco nivelesDesmoldados")
                        
                        time.sleep(10)
                    
                    # Activar finFinalizado mientras nivelesDesmoldados sigue activo
                    if fin_finalizado_var:
                        fin_finalizado_var.set_value(True)
                        print("✅ Ciclo completado correctamente - Activando flanco finFinalizado")
                        
                        # Mantener ambos flancos activos simultáneamente
                        time.sleep(10)
                        
                        # Desactivar finFinalizado primero
                        fin_finalizado_var.set_value(False)
                        print("Flanco finFinalizado desactivado")
                        
                        # Esperar un momento y luego desactivar nivelesDesmoldados
                        time.sleep(10)
                        
                        # Ahora desactivar nivelesDesmoldados
                        if niveles_desmoldados_var:
                            niveles_desmoldados_var.set_value(False)
                            print(f"Flanco nivelesDesmoldados desactivado")
                
                # Tiempo adicional de procesamiento para que la API registre los cambios
                time.sleep(12)
            
            idReceta_actual = (idReceta_actual % len(RecetaNombres)) + 1
            print(f"\n Preparando siguiente receta: {idReceta_actual}- {RecetaNombres[idReceta_actual - 1]}")

            dato_ciclo = None
            for var in variables:
                if var.get_browse_name().Name == "Ciclo_iniciado":
                    dato_ciclo = var
                    break
            
            if dato_ciclo:
                dato_ciclo.set_value(False)
            
            time.sleep(5)
            print(f"\n✅ Ciclo de desmoldeo completado para torre {torre_actual}")
            
            # Actualizar valores para el siguiente ciclo
            gripper_actual = (gripper_actual % 4) + 1
            torre_actual += 1
            torre_proxima = torre_actual + 1
            if torre_actual > 80:
                torre_actual = 1
                torre_proxima = 2

        time.sleep(15)

if __name__ == "__main__":

        ciclo_de_desmoldeo()
